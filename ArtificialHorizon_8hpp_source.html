71 bool on_draw(const Cairo::RefPtr<Cairo::Context>& cr) override {
72 const int w = get_allocated_width();
73 const int h = get_allocated_height();
74
75 // Layout
76 const double readout_h = 34.0; // Space for numeric readouts at bottom
77 const double cx = w / 2.0;
78 const double cy = (h - readout_h) / 2.0;
79 const double radius = std::min(cx, cy) - 6.0;
80
81 bool roll_warning = std::abs(roll_deg_) > roll_warn_;
82 bool pitch_warning = std::abs(pitch_deg_) > pitch_warn_;
83 bool any_warning = roll_warning || pitch_warning;
84
85 double roll_rad = roll_deg_ * M_PI / 180.0;
86
87 // Pitch drives vertical offset of the horizon.
88 // Scale: at pitch_warn_ degrees, the horizon reaches the edge of the circle.
89 double pitch_pixels_per_deg = (radius * 0.7) / pitch_warn_;
90 double pitch_offset = -pitch_deg_ * pitch_pixels_per_deg;
91
92 // --- Outer ring ---
93 cr->set_line_width(2.0);
94 if (is_light_mode_) {
95 cr->set_source_rgba(0.55, 0.55, 0.55, 0.7);
96 } else {
97 cr->set_source_rgba(0.22, 0.30, 0.35, 0.9);
98 }
99 cr->arc(cx, cy, radius, 0, 2 * M_PI);
100 cr->stroke();
101
102 // --- Sky/ground fill (clipped to circle, rotated + shifted) ---
103 cr->save();
104 cr->arc(cx, cy, radius - 1, 0, 2 * M_PI);
105 cr->clip();
106
107 cr->save();
108 cr->translate(cx, cy);
109 cr->rotate(roll_rad);
110
111 // Sky
112 cr->set_source_rgba(0.15, 0.35, 0.65, 0.40);
113 cr->rectangle(-radius * 2, -radius * 2, radius * 4, radius * 2 + pitch_offset);
114 cr->fill();
115
116 // Ground
117 cr->set_source_rgba(0.40, 0.25, 0.12, 0.35);
118 cr->rectangle(-radius * 2, pitch_offset, radius * 4, radius * 2);
119 cr->fill();
120
121 // --- Horizon line ---
122 {
123 double r, g, b;
124 if (any_warning) {
125 r = 0.94; g = 0.27; b = 0.27;
126 } else {
127 double ratio = std::max(std::abs(roll_deg_) / roll_warn_,
128 std::abs(pitch_deg_) / pitch_warn_);
129 ratio = std::min(ratio, 1.0);
130 if (ratio > 0.7) {
131 double t = (ratio - 0.7) / 0.3;
132 r = 0.98 * t + 0.38 * (1 - t);
133 g = 0.75 * t + 0.65 * (1 - t);
134 b = 0.17 * t + 0.98 * (1 - t);
135 } else {
136 r = 0.38; g = 0.65; b = 0.98;
137 }
138 }
139
140 cr->set_source_rgb(r, g, b);
141 cr->set_line_width(2.5);
142 cr->move_to(-radius * 1.5, pitch_offset);
143 cr->line_to(radius * 1.5, pitch_offset);
144 cr->stroke();
145 }
146
147 // --- Pitch ladder lines (every 10° above and below horizon) ---
148 {
149 cr->set_line_width(1.0);
150 double ladder_half_w = radius * 0.25;
151 for (int deg = -40; deg <= 40; deg += 10) {
152 if (deg == 0) continue;
153 double y = pitch_offset - deg * pitch_pixels_per_deg;
154
155 // Only draw if visible within the circle
156 if (std::abs(y) > radius * 0.85) continue;
157
158 if (is_light_mode_) {
159 cr->set_source_rgba(0.3, 0.3, 0.3, 0.5);
160 } else {
161 cr->set_source_rgba(0.7, 0.7, 0.7, 0.4);
162 }
163
164 // Dashed for negative pitch, solid for positive
165 if (deg < 0) {
166 std::vector<double> dashes = {4.0, 3.0};
167 cr->set_dash(dashes, 0);
168 }
169 cr->move_to(-ladder_half_w, y);
170 cr->line_to(ladder_half_w, y);
171 cr->stroke();
172 cr->unset_dash();
173
174 // Small degree label
175 cr->select_font_face("monospace", Cairo::FONT_SLANT_NORMAL, Cairo::FONT_WEIGHT_NORMAL);
176 cr->set_font_size(8.0);
177 char buf[8];
178 std::snprintf(buf, sizeof(buf), "%d", deg);
179 cr->move_to(ladder_half_w + 4, y + 3);
180 cr->show_text(buf);
181 }
182 }
183
184 cr->restore(); // Undo rotate+translate
185 cr->reset_clip();
186 cr->restore(); // Undo clip
187
188 // --- Roll tick marks around the outer ring ---
189 {
190 cr->set_line_width(1.0);
191 double tick_inner = radius - 6;
192 double tick_outer = radius - 1;
193
194 for (int deg = -60; deg <= 60; deg += 10) {
195 if (deg == 0) continue;
196 double a = (deg - 90) * M_PI / 180.0;
197
198 bool is_warn_tick = (std::abs(deg) >= (int)roll_warn_);
199 if (is_warn_tick) {
200 cr->set_source_rgba(0.94, 0.27, 0.27, 0.5);
201 } else {
202 if (is_light_mode_) cr->set_source_rgba(0.5, 0.5, 0.5, 0.4);
203 else cr->set_source_rgba(0.4, 0.48, 0.53, 0.4);
204 }
205
206 double inner = (deg % 30 == 0) ? tick_inner - 4 : tick_inner;
207 cr->move_to(cx + inner * std::cos(a), cy + inner * std::sin(a));
208 cr->line_to(cx + tick_outer * std::cos(a), cy + tick_outer * std::sin(a));
209 cr->stroke();
210 }
211 }
212
213 // --- Roll pointer (small triangle at top, rotates with roll) ---
214 {
215 cr->save();
216 cr->translate(cx, cy);
217 cr->rotate(roll_rad);
218
219 double ptr_y = -(radius - 10);
220 cr->set_source_rgb(1.0, 1.0, 1.0);
221 cr->move_to(0, ptr_y);
222 cr->line_to(-5, ptr_y - 8);
223 cr->line_to(5, ptr_y - 8);
224 cr->close_path();
225 cr->fill();
226
227 cr->restore();
228 }
229
230 // --- Fixed reference marker (top center, stationary) ---
231 {
232 double top_y = cy - radius + 10;
233 if (is_light_mode_) cr->set_source_rgba(0.3, 0.3, 0.3, 0.6);
234 else cr->set_source_rgba(0.6, 0.6, 0.6, 0.6);
235 cr->move_to(cx, top_y);
236 cr->line_to(cx - 5, top_y - 7);
237 cr->line_to(cx + 5, top_y - 7);
238 cr->close_path();
239 cr->stroke();
240 }
241
242 // --- Fixed wing markers (center, green) ---
243 {
244 cr->set_source_rgb(0.29, 0.85, 0.50);
245 cr->set_line_width(2.5);
246
247 // Left wing
248 cr->move_to(cx - radius + 14, cy);
249 cr->line_to(cx - radius * 0.35, cy);
250 cr->stroke();
251 // Left wing down-tick
252 cr->move_to(cx - radius * 0.35, cy);
253 cr->line_to(cx - radius * 0.35, cy + 6);
254 cr->stroke();
255
256 // Right wing
257 cr->move_to(cx + radius - 14, cy);
258 cr->line_to(cx + radius * 0.35, cy);
259 cr->stroke();
260 // Right wing down-tick
261 cr->move_to(cx + radius * 0.35, cy);
262 cr->line_to(cx + radius * 0.35, cy + 6);
263 cr->stroke();
264
265 // Center dot
266 cr->arc(cx, cy, 3, 0, 2 * M_PI);
267 cr->fill();
268 }
269
270 // --- Numeric readouts ---
271 double readout_y = cy + radius + 16;
272
273 cr->select_font_face("monospace", Cairo::FONT_SLANT_NORMAL, Cairo::FONT_WEIGHT_BOLD);
274 cr->set_font_size(12.0);
275
276 // Roll readout (left side)
277 {
278 char buf[24];
279 std::snprintf(buf, sizeof(buf), "R %.1f\xC2\xB0", roll_deg_);
280 std::string text(buf);
281
282 if (roll_warning) cr->set_source_rgb(0.94, 0.27, 0.27);
283 else set_text_color(cr);
284
285 Cairo::TextExtents te;
286 cr->get_text_extents(text, te);
287 cr->move_to(cx - radius * 0.5 - te.width / 2.0, readout_y);
288 cr->show_text(text);
289 }
290
291 // Pitch readout (right side)
292 {
293 char buf[24];
294 std::snprintf(buf, sizeof(buf), "P %.1f\xC2\xB0", pitch_deg_);
295 std::string text(buf);
296
297 if (pitch_warning) cr->set_source_rgb(0.94, 0.27, 0.27);
298 else set_text_color(cr);
299
300 Cairo::TextExtents te;
301 cr->get_text_extents(text, te);
302 cr->move_to(cx + radius * 0.5 - te.width / 2.0, readout_y);
303 cr->show_text(text);
304 }
305
306 // --- Label ---
307 cr->set_font_size(10.0);
308 if (is_light_mode_) cr->set_source_rgba(0.3, 0.3, 0.3, 0.8);
309 else cr->set_source_rgba(0.55, 0.63, 0.67, 0.8);
310 Cairo::TextExtents lte;
311 cr->get_text_extents(label_, lte);
312 cr->move_to(cx - lte.width / 2.0, readout_y + 14);
313 cr->show_text(label_);
314
315 return true;
316 }

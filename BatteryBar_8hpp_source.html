70 current_v = std::min(current_v, kv.second);

82 const double bar_w = w - bar_x - readout_w - warn_w - pad_x - 16;
83 const double bar_y = (h - 14) / 2.0;
84 const double bar_h = 14.0;
85
86 if (bar_w < 20) return true; // Not enough space

88 // "BAT" label
89 cr->select_font_face("monospace", Cairo::FONT_SLANT_NORMAL, Cairo::FONT_WEIGHT_BOLD);
90 cr->set_font_size(10.0);
91 if (is_light_mode_) cr->set_source_rgba(0.3, 0.3, 0.3, 0.8);
92 else cr->set_source_rgba(0.55, 0.63, 0.67, 0.8);
93 cr->move_to(pad_x, h / 2.0 + 4);
94 cr->show_text("BAT");
95
96 // Bar track
97 rounded_rect(cr, bar_x, bar_y, bar_w, bar_h, 3.0);
98 if (is_light_mode_) {

100 } else {
101 cr->set_source_rgba(0.10, 0.16, 0.20, 0.9);
102 }
103 cr->fill_preserve();
104 if (is_light_mode_) cr->set_source_rgba(0.7, 0.7, 0.7, 0.6);
105 else cr->set_source_rgba(0.20, 0.28, 0.33, 0.6);
106 cr->set_line_width(1.0);
107 cr->stroke();
108
109 // Fill bar
110 if (has_data) {
111 double ratio = std::clamp((double)(current_v - min_voltage_) / (max_voltage_ - min_voltage_), 0.0, 1.0);
112 double fill_w = ratio * (bar_w - 4);
113
114 double r, g, b;
115 get_voltage_color(current_v, r, g, b);
116
117 if (fill_w > 1) {

120 cr->fill();
121 }
122
123 // Warning threshold marker line on the bar
124 double warn_ratio = std::clamp((double)(warning_voltage_ - min_voltage_) / (max_voltage_ - min_voltage_), 0.0, 1.0);
125 double warn_x = bar_x + 2 + warn_ratio * (bar_w - 4);
126 cr->set_source_rgba(0.98, 0.75, 0.17, 0.5);

135 cr->set_source_rgb(r, g, b);

142 } else {

191 void rounded_rect(const Cairo::RefPtr<Cairo::Context>& cr,
192 double x, double y, double w, double h, double r) {
193 cr->begin_new_sub_path();
194 cr->arc(x + w - r, y + r, r, -M_PI / 2.0, 0);

196 cr->arc(x + r, y + h - r, r, M_PI / 2.0, M_PI);
197 cr->arc(x + r, y + r, r, M_PI, 3.0 * M_PI / 2.0);

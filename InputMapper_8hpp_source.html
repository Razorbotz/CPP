176 // ── Runtime Remapping ──────────────────────────────────────────────────

178 // Build a fast lookup table: key = (inputJoystick << 8 | inputAxis)
179 // value = (outputJoystick, outputAxis, invert)
180 struct RemapResult {
181 uint8_t which;
182 uint8_t axis;
183 bool invert;
184 };
185
186 std::map<int, RemapResult> buildRemapTable() const {
187 std::map<int, RemapResult> table;
188 for (auto& m : axisMappings) {
189 int key = (m.inputJoystick << 8) | m.inputAxis;
190 table[key] = { (uint8_t)m.outputJoystick, (uint8_t)m.outputAxis, m.invert };
191 }
192 return table;
193 }
194
195 // Direct remap function — replaces remapJoystickInputs()
196 // Returns true if a mapping was found (which/axis are modified in-place).
197 // Returns false if no mapping exists (which/axis are unchanged — passthrough).
198 bool remap(uint8_t* which, uint8_t* axis, bool* invert = nullptr) const {
199 int key = (*which << 8) | *axis;
200 auto it = remapTable_.find(key);
201 if (it != remapTable_.end()) {
202 *which = it->second.which;
203 *axis = it->second.axis;
204 if (invert) *invert = it->second.invert;
205 return true;
206 }
207 return false;
208 }
209
210 // Call this after loading to pre-build the lookup table
211 void buildLookup() {
212 remapTable_ = buildRemapTable();
213 }
214
215private:
216 std::map<int, RemapResult> remapTable_;
217};

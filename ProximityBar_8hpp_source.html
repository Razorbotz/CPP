127 bool on_draw(const Cairo::RefPtr<Cairo::Context>& cr) override {
128 // Fully faded out — draw nothing (transparent, video shows through)
129 if (fade_alpha_ < 0.01) return true;
130
131 const int w = get_allocated_width();
132 const int h = get_allocated_height();
133
134 // Push a group so the entire widget fades uniformly during transitions
135 cr->push_group();
136
137 // --- Layout constants ---
138 const double pad_top = 26.0;
139 const double pad_bottom = 62.0;
140 const double bar_w = std::min((double)w * 0.38, 38.0);
141 const double bar_x = (w - bar_w) / 2.0 - 18.0;
142 const double bar_top = pad_top;
143 const double bar_h = h - pad_top - pad_bottom;
144 const double bar_bottom = bar_top + bar_h;
145 const double tick_x = bar_x + bar_w + 8.0;
146
147 // --- Semi-transparent background panel ---
148 // Matches the transparent overlay approach used throughout the GUI.
149 // The edge panels use "rgba(..., 0.0)" CSS backgrounds so video
150 // shows through; here we use a slight tint so the bar is readable
151 // over any video content without fully occluding the feed.
152 rounded_rect(cr, 0, 0, w, h, 8.0);
153 if (is_light_mode_) {
154 cr->set_source_rgba(0.94, 0.98, 0.95, 0.55);
155 } else {
156 cr->set_source_rgba(0.043, 0.102, 0.129, 0.55);
157 }
158 cr->fill();
159
160 // --- "PROXIMITY" label ---
161 cr->select_font_face("monospace", Cairo::FONT_SLANT_NORMAL, Cairo::FONT_WEIGHT_BOLD);
162 cr->set_font_size(10.0);
163 set_text_color(cr, 0.55, 0.63, 0.67, 0.3, 0.3, 0.3);
164 Cairo::TextExtents te;
165 cr->get_text_extents("PROXIMITY", te);
166 cr->move_to((w - te.width) / 2.0, 17.0);
167 cr->show_text("PROXIMITY");
168
169 // --- Bar track ---
170 rounded_rect(cr, bar_x, bar_top, bar_w, bar_h, 5.0);
171 if (is_light_mode_) {
172 cr->set_source_rgba(0.78, 0.78, 0.78, 0.8);
173 } else {
174 cr->set_source_rgba(0.086, 0.125, 0.157, 0.9);
175 }
176 cr->fill_preserve();
177 if (is_light_mode_) {
178 cr->set_source_rgba(0.6, 0.6, 0.6, 0.8);
179 } else {
180 cr->set_source_rgba(0.165, 0.227, 0.267, 0.9);
181 }
182 cr->set_line_width(1.0);
183 cr->stroke();
184
185 // --- Zone geometry ---
186 // Top of bar = 0m (closest), bottom = max_distance (farthest)
187 auto dist_to_y = [&](double d) -> double {
188 return bar_top + (1.0 - d / max_distance_m_) * bar_h;
189 };
190
191 double warn_y = dist_to_y(warning_threshold_m_);
192 double opt_y = dist_to_y(optimal_threshold_m_);
193 Zone current_zone = get_current_zone();
194
195 double active_alpha = 0.90;
196 double dim_alpha = 0.18;
197
198 // TOP zone: too far (red)
199 // from max distance down to optimal threshold
200 draw_zone(cr, bar_x + 2, bar_top + 2, bar_w - 4, opt_y - bar_top - 2,
201 0.86, 0.15, 0.15,
202 current_zone == Zone::APPROACH ? active_alpha : dim_alpha);
203
204 // MIDDLE zone: optimal (green)
205 // from optimal threshold down to warning threshold
206 draw_zone(cr, bar_x + 2, opt_y, bar_w - 4, warn_y - opt_y,
207 0.133, 0.773, 0.369,
208 current_zone == Zone::OPTIMAL ? active_alpha : dim_alpha);
209
210 // BOTTOM zone: too close (red)
211 // from warning threshold down to 0.0m
212 draw_zone(cr, bar_x + 2, warn_y, bar_w - 4, bar_bottom - 2 - warn_y,
213 0.86, 0.15, 0.15,
214 current_zone == Zone::WARNING ? active_alpha : dim_alpha);
215
216 // --- Needle ---
217 {
218 double needle_y = dist_to_y(distance_m_);
219 needle_y = std::max(bar_top + 2.0, std::min(bar_bottom - 2.0, needle_y));
220
221 cr->set_source_rgb(1.0, 1.0, 1.0);
222 cr->set_line_width(2.5);
223 cr->move_to(bar_x + 2, needle_y);
224 cr->line_to(bar_x + bar_w - 2, needle_y);
225 cr->stroke();
226 }
227
228 // --- Tick labels ---
229 cr->select_font_face("monospace", Cairo::FONT_SLANT_NORMAL, Cairo::FONT_WEIGHT_NORMAL);
230 cr->set_font_size(10.0);
231
232 double dim_tr, dim_tg, dim_tb;
233 get_dim_tick_color(dim_tr, dim_tg, dim_tb);
234
235 draw_tick(cr, dist_to_y(0.0), tick_x, bar_x + bar_w, "0.0m", dim_tr, dim_tg, dim_tb);
236 draw_tick(cr, dist_to_y(warning_threshold_m_), tick_x, bar_x + bar_w,
237 format_dist(warning_threshold_m_), 0.86, 0.15, 0.15);
238 draw_tick(cr, dist_to_y(optimal_threshold_m_), tick_x, bar_x + bar_w,
239 format_dist(optimal_threshold_m_), 0.133, 0.773, 0.369);
240 draw_tick(cr, dist_to_y(max_distance_m_), tick_x, bar_x + bar_w,
241 format_dist(max_distance_m_), 0.86, 0.15, 0.15);
242
243 // --- Distance readout ---
244 cr->select_font_face("monospace", Cairo::FONT_SLANT_NORMAL, Cairo::FONT_WEIGHT_BOLD);
245 cr->set_font_size(20.0);
246 if (current_zone == Zone::WARNING) {
247 // Blink the distance text red when too close
248 double text_alpha = 1.0;
249 cr->set_source_rgba(0.94, 0.27, 0.27, text_alpha);
250 } else {
251 set_text_color(cr, 1.0, 1.0, 1.0, 0.0, 0.0, 0.0);
252 }
253 std::string dist_text = format_dist(distance_m_);
254 cr->get_text_extents(dist_text, te);
255 cr->move_to((w - te.width) / 2.0, bar_bottom + 24);
256 cr->show_text(dist_text);
257
258 // --- Status badge (blinks only for WARNING) ---
259 draw_status_badge(cr, w, bar_bottom + 32, current_zone);
260
261 // Pop group and paint with fade alpha
262 cr->pop_group_to_source();
263 cr->paint_with_alpha(fade_alpha_);
264
265 return true;
266 }

285 // --- Timers ---
286
287 bool on_blink_tick() {
288 if (get_current_zone() == Zone::WARNING && fade_alpha_ > 0.01) {
289 blink_on_ = !blink_on_;
290 queue_draw();
291 } else {
292 blink_on_ = true;
293 }
294 return true;
295 }
296
297 bool on_fade_tick() {
298 double target = bar_visible_ ? 1.0 : 0.0;
299 double step = 0.1; // ~330ms full transition at 33ms ticks
300
301 if (std::abs(fade_alpha_ - target) < step) {
302 if (fade_alpha_ != target) {
303 fade_alpha_ = target;
304 queue_draw();
305 }
306 } else {
307 fade_alpha_ += (target > fade_alpha_) ? step : -step;
308 queue_draw();
309 }
310 return true;
311 }

360 void draw_status_badge(const Cairo::RefPtr<Cairo::Context>& cr,
361 double container_w, double badge_y, Zone zone) {
362 std::string text;
363 double bg_r, bg_g, bg_b, fg_r, fg_g, fg_b;
364
365 switch (zone) {
366 case Zone::WARNING:
367 text = "TOO CLOSE";
368 bg_r = 0.50; bg_g = 0.11; bg_b = 0.11;
369 fg_r = 0.97; fg_g = 0.44; fg_b = 0.44;
370 if (!blink_on_) {
371 bg_r *= 0.4; bg_g *= 0.4; bg_b *= 0.4;
372 }
373 break;
374 case Zone::OPTIMAL:
375 text = "OPTIMAL";
376 bg_r = 0.08; bg_g = 0.33; bg_b = 0.18;
377 fg_r = 0.29; fg_g = 0.85; fg_b = 0.50;
378 break;
379 case Zone::APPROACH:
380 text = "TOO FAR";
381 bg_r = 0.50; bg_g = 0.11; bg_b = 0.11;
382 fg_r = 0.97; fg_g = 0.44; fg_b = 0.44;
383 break;
384 }
385
386 cr->select_font_face("monospace", Cairo::FONT_SLANT_NORMAL, Cairo::FONT_WEIGHT_BOLD);
387 cr->set_font_size(11.0);
388 Cairo::TextExtents te;
389 cr->get_text_extents(text, te);
390 double bw = te.width + 18;
391 double bh = 22;
392 double bx = (container_w - bw) / 2.0;
393
394 cr->set_source_rgb(bg_r, bg_g, bg_b);
395 rounded_rect(cr, bx, badge_y, bw, bh, 4.0);
396 cr->fill();
397
398 cr->set_source_rgb(fg_r, fg_g, fg_b);
399 cr->move_to(bx + 9, badge_y + 15.5);
400 cr->show_text(text);
401 }

403 void rounded_rect(const Cairo::RefPtr<Cairo::Context>& cr,
404 double x, double y, double w, double h, double r) {
405 cr->begin_new_sub_path();
406 cr->arc(x + w - r, y + r, r, -M_PI / 2.0, 0);
407 cr->arc(x + w - r, y + h - r, r, 0, M_PI / 2.0);
408 cr->arc(x + r, y + h - r, r, M_PI / 2.0, M_PI);
409 cr->arc(x + r, y + r, r, M_PI, 3.0 * M_PI / 2.0);
410 cr->close_path();
411 }
